{{- /* Links in docs/*.md are written for GitHub. Turn them into site links: - `concepts.md#baseline` → the Concepts page; `README.md` → the docs section. A link to a page that doesn't exist fails the build. - `../examples/…`, `../CONTRIBUTING.md`: files outside docs/ → GitHub. Everything else is left as written. */ -}} {{- $dest := .Destination -}} {{- $external := false -}} {{- with try (urls.Parse .Destination) -}} {{- with .Value -}} {{- $fragment := cond (ne .Fragment "") (printf "#%s" .Fragment) "" -}} {{- if .IsAbs -}} {{- $external = true -}} {{- else if hasPrefix .Path "../" -}} {{- $dest = printf "https://github.com/packtrix/packtrix/blob/main/%s%s" (strings.TrimPrefix "../" .Path) $fragment -}} {{- $external = true -}} {{- else if strings.HasSuffix .Path ".md" -}} {{- $name := strings.TrimSuffix ".md" (path.Base .Path) -}} {{- $target := cond (eq $name "README") "/docs" (printf "/docs/%s" $name) -}} {{- with site.GetPage $target -}} {{- $dest = printf "%s%s" .RelPermalink $fragment -}} {{- else -}} {{- errorf "%s links to %q, which is not a docs page" $.Page.Path $.Destination -}} {{- end -}} {{- else if hasPrefix .Path "/" -}} {{- $dest = relURL (strings.TrimPrefix "/" $.Destination) -}} {{- end -}} {{- end -}} {{- end -}} {{ .Text }} {{- /* chomp trailing newline */ -}}
